# One-time check before your first commit in each repo. From the repo root:
#
#   PowerShell:
#     git ls-files | Select-String -Pattern '^(CLAUDE\.md|\.claude/|\.mcp\.json)' |
#       ForEach-Object { $_.Line } |
#       ForEach-Object { Select-String -Path $_ -Pattern '[A-Za-z]:\\' }
#
#   Git Bash:
#     git ls-files 'CLAUDE.md' '.claude/*' '.mcp.json' | xargs grep -n '[A-Za-z]:\\'
#
# Any hit is a machine-specific path in a file you are about to commit.
# Move it to CLAUDE.local.md or .claude/settings.local.json.
#
# Also worth checking for secrets before the first commit of .mcp.json:
#     git ls-files '.mcp.json' | xargs grep -niE 'token|secret|key|password'
